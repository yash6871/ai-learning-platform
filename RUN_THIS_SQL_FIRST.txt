IMPORTANT — run this in Supabase's SQL Editor BEFORE deploying the backend.

create_tables.py only creates missing TABLES, not missing COLUMNS on tables
that already exist. Without running this, the backend will crash with a
"column does not exist" error.

-- New: custom termination threshold per assessment (default 10)
ALTER TABLE assessments ADD COLUMN IF NOT EXISTS max_violations INTEGER DEFAULT 10;

-- New: last violation reason/severity, shown to faculty in Monitor
ALTER TABLE results ADD COLUMN IF NOT EXISTS last_violation_reason VARCHAR(200);
ALTER TABLE results ADD COLUMN IF NOT EXISTS last_violation_severity VARCHAR(20);

-- (Older migrations, keep if not already applied — safe to re-run, IF NOT EXISTS)
ALTER TABLE mock_interview_evaluation ADD COLUMN IF NOT EXISTS max_score FLOAT DEFAULT 100;
ALTER TABLE assessments ADD COLUMN IF NOT EXISTS active_from TIMESTAMPTZ;
ALTER TABLE assessments ADD COLUMN IF NOT EXISTS active_until TIMESTAMPTZ;
